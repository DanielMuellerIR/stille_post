cmake_minimum_required(VERSION 3.22)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" STILLEPOST_VERSION LIMIT_COUNT 1)
project(StillePostLinux VERSION ${STILLEPOST_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Multimedia DBus Network)
find_package(X11 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
pkg_check_modules(QRENCODE REQUIRED IMPORTED_TARGET libqrencode)
find_package(OpenSSL REQUIRED)
find_package(Qt6 6.4 REQUIRED COMPONENTS WaylandClient)
find_package(LayerShellQt REQUIRED)
add_library(stillepost-shell MODULE wayland_shell.cpp)
target_include_directories(stillepost-shell PRIVATE /opt/stillepost-layershell-src/src /opt/stillepost-layershell-build/src/LayerShellQt)
target_link_libraries(stillepost-shell PRIVATE Qt6::WaylandClientPrivate LayerShellQt::Interface)
set_target_properties(stillepost-shell PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/wayland-shell-integration")
add_executable(stillepost main.cpp shortcut_codec.cpp app_controller.cpp desktop_placement.cpp audio_import_window.cpp test_dictation_window.cpp hotkeys.cpp x11_hotkeys.cpp ui_common.cpp settings_window.cpp history_window.cpp accessibility.cpp audio_input.cpp)
add_dependencies(stillepost stillepost-shell)
target_link_libraries(stillepost PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::DBus Qt6::Network X11::X11 ${X11_Xtst_LIB} OpenSSL::Crypto PkgConfig::ATSPI PkgConfig::PULSE PkgConfig::QRENCODE LayerShellQt::Interface)
target_compile_definitions(stillepost PRIVATE STILLEPOST_VERSION="${PROJECT_VERSION}")
install(TARGETS stillepost RUNTIME DESTINATION bin)
option(STILLEPOST_BUILD_DESKTOP_TESTS "Isolierte Desktopprüfungen bauen" OFF)
if(STILLEPOST_BUILD_DESKTOP_TESTS)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  add_executable(stillepost-portal-test tests/portal_test.cpp hotkeys.cpp)
  target_include_directories(stillepost-portal-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(stillepost-portal-test PRIVATE Qt6::Core Qt6::DBus Qt6::Test)
  add_executable(stillepost-hotkey-test tests/hotkey_test.cpp x11_hotkeys.cpp)
  target_include_directories(stillepost-hotkey-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(stillepost-hotkey-test PRIVATE Qt6::Widgets Qt6::Test X11::X11)
  add_executable(stillepost-accessibility-test tests/accessibility_test.cpp accessibility.cpp)
  target_include_directories(stillepost-accessibility-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(stillepost-accessibility-test PRIVATE Qt6::Widgets Qt6::DBus Qt6::Test PkgConfig::ATSPI)
  add_executable(stillepost-window-test tests/window_test.cpp app_controller.cpp desktop_placement.cpp audio_import_window.cpp test_dictation_window.cpp shortcut_codec.cpp hotkeys.cpp x11_hotkeys.cpp ui_common.cpp settings_window.cpp history_window.cpp accessibility.cpp audio_input.cpp)
  target_include_directories(stillepost-window-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(stillepost-window-test PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::DBus Qt6::Network Qt6::Test X11::X11 ${X11_Xtst_LIB} OpenSSL::Crypto PkgConfig::ATSPI PkgConfig::PULSE PkgConfig::QRENCODE LayerShellQt::Interface)

endif()
